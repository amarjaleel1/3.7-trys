# Simple program demonstrating basic operations
# This program calculates a simple expression and stores it in memory

# Initialize values
ADD R1, R0, 10    # R1 = 10
ADD R2, R0, 5     # R2 = 5
ADD R3, R0, 2     # R3 = 2

# Perform calculations
MUL R4, R1, R2    # R4 = R1 * R2 = 50
DIV R5, R4, R3    # R5 = R4 / R3 = 25
SUB R6, R4, R5    # R6 = R4 - R5 = 25

# Store result in memory
SW R6, 100(R0)    # Mem[100] = R6 = 25

# Load result back to verify
LW R7, 100(R0)    # R7 = Mem[100] = 25

# Conditional branch
BEQ R6, R7, 3     # If R6 == R7, skip 3 instructions
ADD R8, R0, 1     # This should be skipped
ADD R9, R0, 2     # This should be skipped
JUMP 15           # This should be skipped

# Final result confirmation
ADD R10, R7, R0   # R10 = R7 = 25
